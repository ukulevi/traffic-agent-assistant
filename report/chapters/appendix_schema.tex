\section{Domain Schemas}

\begin{lstlisting}[caption={Các schema cốt lõi}]
IncidentVector:
  event_type: accident|flood|lane_closure|demand_surge|signal_change
  affected_node_ids: list[str]  # exactly one node in MVP
  severity: low|medium|high
  duration_minutes: int[1,180]
  description: str[1,1000]  # never selects simulation behavior
  lane_closure_ratio: float[0,1]|null
  demand_multiplier: float(1,3]|null
  signal_plan_delta: SignalPlanDelta|null
      green_time_ratio_delta: float[-1,1]
# signal offset deferred beyond MVP
# exactly one event-specific parameter:
# lane_closure -> lane_closure_ratio
# demand_surge -> demand_multiplier
# signal_change -> signal_plan_delta
# accident/flood -> none

WhatIfJobRequest:
  incident: IncidentVector|null
  candidate_action: CandidateAction  # required, non-executable hypothesis
  node_ids: list[str]  # production server-authorized scope

RouteCandidate:
  route_id: str
  topology_version: str
  boundary_entry_node: str
  boundary_exit_node: str
  node_sequence: list[str]  # directed, simple, incident-avoiding
  edge_ids: list[str]       # exactly len(node_sequence)-1
  base_cost: float>0
  distance_m: float>0

RouteEvaluation:
  route: RouteCandidate
  max_vc_ratio: float
  avg_speed_kmh: float
  delay_proxy_seconds: float
  uncertainty_score: float
  ood_score: float
  passed: bool
  rejection_reasons: list[str]
  evidence_complete: bool
  model_version: str
  data_version: str
  topology_version: str

RouteRecommendation:
  route: RouteCandidate
  evaluation: RouteEvaluation  # must pass with complete evidence
  rank: int[1,3]
  executable: false
  requires_operator_approval: true
  applied_by_system: false

SimulationQuery:
  job_id: str
  node_ids: list[str]
  metrics: enum[]
  horizons_minutes: int[]
  aggregation: enum
  limit: int[1,1000]

Citation:
  document_id: str
  document_number: str
  provision: str
  source_url: str
  effective_from: date
  effective_to: date|null
  content_hash: str
\end{lstlisting}
