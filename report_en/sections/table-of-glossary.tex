\chapter*{Glossary of Technical Terms}
\addcontentsline{toc}{chapter}{Glossary of Technical Terms}

\begin{longtable}{|p{4.2cm}|p{10.8cm}|}
\hline
\rowcolor{blue!10}
\textbf{Term} & \textbf{Technical Meaning and Context in STWI} \\
\hline
\endhead

\textbf{Candidate Action} & A synthesized traffic management proposal generated by the agent workflow that has not yet completed full safety and legal verification. \\
\hline
\textbf{Counterfactual Reasoning} & Estimating hypothetical outcomes under alternative interventions (``What would happen to congestion if green ratio increased to 0.85?''). \\
\hline
\textbf{Decision Support Only} & The core architectural policy ensuring STWI produces non-executable advisory recommendations without direct physical device actuation. \\
\hline
\textbf{Fail-Closed} & A safety design principle where any failure, uncertainty, or missing legal grounding halts automated action and escalates to human operator review. \\
\hline
\textbf{Missing Mask ($M$)} & A binary tensor indicator indicating whether a specific node sensor metric is absent at a given 5-minute timestep. \\
\hline
\textbf{Out-of-Distribution (OOD)} & Traffic states or incident vectors outside the bounded feature distribution of the calibrated surrogate training corpus. \\
\hline
\textbf{Recommended Action} & A thoroughly verified traffic signal or diversion plan that passed legal grounding, uncertainty gates, and $V/C \le 0.90$ constraints, awaiting operator confirmation. \\
\hline
\textbf{Surrogate Model} & A lightweight machine learning model (e.g., LightGBM / MLP ensemble) trained on offline SUMO simulation data to achieve sub-second inference. \\
\hline
\textbf{Trace ID} & An immutable UUID assigned at job ingestion that correlates logs, vector queries, surrogate inferences, and audit trails. \\
\hline
\textbf{Volume-to-Capacity (V/C)} & The ratio between actual/predicted traffic volume and the maximum design throughput of an intersection approach. \\
\hline
\textbf{Zero Raw-Video Retention} & Privacy and compliance rule enforcing that raw video feeds are processed at the edge in volatile memory and discarded immediately after aggregate feature extraction. \\
\hline
\end{longtable}
